7def energy_aware_prune(model, amount=0.0):
8    """
9    Prune parameters starting from the largest (proxy for energy intensive).
10    No-op if amount <= 0 to preserve original behavior.
11    """
12    if amount <= 0.0:
13        return
14    parameters_to_prune = []
15    for module in model.modules():
16        for name, _ in module.named_parameters(recurse=False):
17            if 'bias' in name:
18                continue
