21def energy_aware_prune(model, prune_fraction=0.2):
22    """
23    Energy-aware pruning helper.
24    - Estimates layer 'energy' as layer.count_params()
25    - Sorts layers by estimated energy (descending)
26    - For each layer, attempts to prune entire output filters (channels) with smallest L1 norm
27    Note: This function mutates model weights only when explicitly called.
28    """
29    # Collect layers with weights and their energy estimate
30    layers_with_weights = []
31    for layer in model.layers:
32        weights = layer.get_weights()
33        if not weights:
34            continue
35        energy = layer.count_params()
36        layers_with_weights.append((energy, layer))
37    # Sort by energy descending
38    layers_with_weights.sort(key=lambda x: x[0], reverse=True)
39    # Determine how many layers to prune based on model size and fraction
40    total_layers = len(layers_with_weights)
41    layers_to_prune = max(1, int(total_layers * prune_fraction)) if total_layers > 0 else 0
42 
43    pruned_info = []
44    for _, layer in layers_with_weights[:layers_to_prune]:
45        orig_weights = layer.get_weights()
46        if not orig_weights:
47            continue
48        new_weights = []
49        pruned = False
50        for w in orig_weights:
51            arr = np.array(w)
52            # Identify possible output-channel axis heuristically:
53            # For kernels like (kh, kw, in_ch, out_ch) -> prune over last axis
54            # For recurrent kernels or biases, skip pruning if 1D
55            if arr.ndim >= 4:
56                # compute L1 norm per output channel (last axis)
57                norms = np.sum(np.abs(arr), axis=tuple(range(arr.ndim-1)))
58                num_filters = norms.shape[0] if norms.shape else 0
59                if num_filters <= 1:
60                    new_weights.append(arr)
61                    continue
62                # Decide how many filters to zero (prune). Prune up to 10% of filters but at least 1.
63                k = max(1, int(num_filters * 0.1))
64                prune_idx = np.argsort(norms)[:k]
65                arr_copy = arr.copy()
66                # Zero-out selected output channels
67                arr_copy[..., prune_idx] = 0
68                new_weights.append(arr_copy)
69                pruned = True
70            elif arr.ndim == 2:
71                # For 2D weights (e.g., dense kernels), prune last axis similarly
72                norms = np.sum(np.abs(arr), axis=0)
73                num_filters = norms.shape[0]
74                if num_filters <= 1:
75                    new_weights.append(arr)
76                    continue
77                k = max(1, int(num_filters * 0.1))
78                prune_idx = np.argsort(norms)[:k]
79                arr_copy = arr.copy()
80                arr_copy[:, prune_idx] = 0
81                new_weights.append(arr_copy)
82                pruned = True
83            else:
84                # Do not prune biases or 1D params
85                new_weights.append(arr)
86        if pruned:

89                pruned_info.append((layer.name, True))
90            except Exception:
91                # If setting weights fails for any layer, skip pruning for that layer
92                pruned_info.append((layer.name, False))
93    return pruned_info
